CarController.lua
LocalScript → StarterPlayerScripts
-- WASD操作・Shiftブースト・Altドリフト・Eアビリティ・サイレン点滅

local Settings = require(ReplicatedStorage:WaitForChild("CarSettings"))
local vConfig  = Settings.Vehicles[vehicleType]

-- 入力マップ
local inputMap = { Forward=false, Backward=false, Left=false, Right=false,
                   Boost=false, Drift=false, Ability=false }

-- Eキーでアビリティ発動 → サーバーに FireAbility イベントを送信
activateAbility = function()
    if abilityCooldown then return end
    abilityCooldown = true
    evAbility:FireServer(vehicleType)
    task.delay(vConfig.AbilityCooldown, function() abilityCooldown = false end)
end

-- Altキーでドリフト（横方向に滑る）
if inputMap.Drift then
    local lateralVel = bv.Velocity - forward * bv.Velocity:Dot(forward)
    bv.Velocity = forward * bv.Velocity:Dot(forward)
              + lateralVel * Settings.DriftFriction
end

-- 特殊車両はサイレン音＋ライト点滅
if vConfig.HasSiren then
    sirenSound:Play()
    -- 0.3秒ごとにLight1/Light2を交互点滅
end
DeathmatchManager.lua
Script → ServerScriptService
-- サーバー側：車生成・衝突ダメージ・アビリティ処理・リスポーン・コイン

-- 衝突ダメージ（速度差でダメージを計算）
body.Touched:Connect(function(hit)
    local speedDiff = math.abs(body.Velocity.Magnitude - hit.Velocity.Magnitude)
    if speedDiff >= Settings.CollisionSpeedThreshold then
        local damage = math.floor(speedDiff * Settings.CollisionDamageMultiplier)
        applyDamage(player, hitPlayer, damage)
    end
end)

-- HP0 → 爆発 → RespawnDelay 秒後にリスポーン
killPlayer = function(killer, victim)
    local explosion = Instance.new("Explosion")
    explosion.Position = body.Position
    explosion.Parent   = workspace
    victimData.car:Destroy()
    task.delay(Settings.RespawnDelay, function() spawnCar(victim, savedType) end)
end

-- アビリティ処理（EMP例）
if vConfig.Ability == "EMP" then
    for _, otherData in pairs(playerData) do
        if (myBody.Position - otherBody.Position).Magnitude <= vConfig.EMPRadius then
            bv.Velocity = Vector3.zero  -- 一時停止
        end
    end
end

LocalScript → StarterGui
-- ゲーム開始時に車種を選択するUI
-- CarSettings.Vehicles のテーブルからボタンを自動生成

for _, vType in ipairs(vehicleOrder) do
    local btn = Instance.new("TextButton")
    btn.Text  = vConfig.DisplayName .. "\nHP: " .. vConfig.HP

    -- Game Pass 限定バッジ
    if vConfig.RequiresGamePass then
        local badge     = Instance.new("TextLabel")
        badge.Text      = "Game Pass 限定"
        badge.Parent    = btn
    end

    btn.MouseButton1Click:Connect(function()
        evSelectVehicle:FireServer(vType)  -- サーバーに通知
        screenGui:Destroy()
    end)
end
HUD.lua
LocalScript → StarterGui
-- HPバー・コイン・スコア・キルフィード・アイテム通知

-- HPバー（比率に応じて色変化）
evUpdateHP.OnClientEvent:Connect(function(hp, maxHP)
    local ratio = hp / maxHP
    hpBar.Size  = UDim2.new(ratio, 0, 1, 0)
    hpBar.BackgroundColor3 = ratio > 0.5 and Color3.fromRGB(50,200,70)
                          or ratio > 0.25 and Color3.fromRGB(220,175,30)
                          or Color3.fromRGB(220,50,50)
end)

-- キルフィード（3秒で自動フェードアウト）
evKilled.OnClientEvent:Connect(function(killerName, victimName)
    killFeedLabel.Text = killerName .. " が " .. victimName .. " を撃破！"
    task.delay(3, function()
        TweenService:Create(killFeedLabel, TweenInfo.new(0.5), { TextTransparency=1 }):Play()
    end)
end)

🔧Roblox Studio セットアップ手順

新規プレースを作成

Roblox Studio を開き、Baseplate または新規プレースを作成してください。

スクリプトを配置

CarSettings.lua → ReplicatedStorage に ModuleScript として配置
CarController.lua → StarterPlayerScripts に LocalScript として配置
DeathmatchManager.lua → ServerScriptService に Script として配置
VehicleSelectGui.lua → StarterGui に LocalScript として配置
HUD.lua → StarterGui に LocalScript として配置
MonetizationManager.lua → ServerScriptService に Script として配置

CarTemplate モデルを用意

車のモデルを作成し、Body という名前の PrimaryPart を設定。 必要に応じて Light1 / Light2（PointLight）と SirenSound（Sound）を子に追加。 モデルを ReplicatedStorage に CarTemplate という名前で配置。

スポーン地点を配置

Workspace に Spawns フォルダを作成し、SpawnLocation を複数配置。

マップを用意（任意）

Workspace に Maps フォルダを作成し、 City / Desert / Factory サブフォルダを配置。 新マップ追加は CarSettings.lua の CarSettings.Maps リストに名前を追記するだけ。

課金IDを差し替え（任意）

MonetizationManager.lua の GAME_PASSES / DEV_PRODUCTS テーブルの ID を、Roblox Creator Dashboard で取得した実際のIDに変更。

テストプレイ

Roblox Studio の「テスト」タブ → 「プレイ」で動作確認。 車種選択GUIが表示され、WASD で移動・Shift でブースト・Alt でドリフト・E でアビリティが使えれば成功。

🔌拡張ガイド

やりたいこと変更場所手順
新しい車種を追加 CarSettings.lua CarSettings.Vehicles テーブルに1エントリ追加
新しいアビリティを追加 DeathmatchManager.lua handleAbility 内に elseif vConfig.Ability == "XXX" を追加
新しいマップを追加 CarSettings.lua CarSettings.Maps に名前を追加し、Workspace/Maps に同名フォルダを作成
新しいアイテムを追加 CarSettings.lua CarSettings.Items テーブルに1エントリ追加
新しいGame Passを追加 MonetizationManager.lua GAME_PASSES テーブルに { id=..., flag=... } を追加
新しい商品（Developer Product）を追加 MonetizationManager.lua DEV_PRODUCTS テーブルに [ID] = function(player) ... end を追加
